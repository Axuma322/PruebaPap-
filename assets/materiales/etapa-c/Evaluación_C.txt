function shuffleArray(arr){
  const a = arr.slice();
  for(let i = a.length - 1; i > 0; i--){
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

// Reordena las opciones de cada pregunta y recalcula el índice de la respuesta correcta.
// Se ejecuta una vez por carga de página, así cada intento presenta un orden distinto.
function shuffleQuestionOptions(list){
  return list.map(item=>{
    const indices = shuffleArray(item.options.map((_, i)=>i));
    const newOptions = indices.map(i=>item.options[i]);
    const newCorrect = indices.indexOf(item.correct);
    return { q:item.q, options:newOptions, correct:newCorrect };
  });
}

const scarfQuestionsShuffled = shuffleQuestionOptions(scarfQuestions);

function buildQuestions(list, container, prefix){
  list.forEach((item, idx)=>{
    const div = document.createElement('div');
    div.className = 'question';
    div.id = prefix + '-' + idx;
    let optsHtml = '';
    item.options.forEach((opt, oidx)=>{
      optsHtml += `<label><input type="radio" name="${prefix}-${idx}" value="${oidx}"> ${opt}</label>`;
    });
    div.innerHTML = `<p class="qtext"><span class="qnum">${idx+1}.</span>${item.q}</p>
      <div class="options">${optsHtml}</div>
      <div class="feedback" id="fb-${prefix}-${idx}"></div>`;
    container.appendChild(div);
  });
}

buildQuestions(scarfQuestionsShuffled, document.getElementById('scarfBlock'), 'scarf');

document.getElementById('submitBtn').addEventListener('click', function(){
  let allAnswered = true;
  let correctCount = 0;

  scarfQuestionsShuffled.forEach((item, idx)=>{
    const selected = document.querySelector(`input[name="scarf-${idx}"]:checked`);
    const qDiv = document.getElementById('scarf-' + idx);
    const fb = document.getElementById('fb-scarf-' + idx);
    if(!selected){
      allAnswered = false;
      qDiv.classList.remove('correct','incorrect');
      fb.classList.remove('show');
      return;
    }
    const val = parseInt(selected.value);
    if(val === item.correct){
      correctCount++;
      qDiv.classList.add('correct');
      qDiv.classList.remove('incorrect');
      fb.textContent = 'Correcto.';
      fb.className = 'feedback show correct';
    } else {
      qDiv.classList.add('incorrect');
      qDiv.classList.remove('correct');
      fb.textContent = 'Incorrecto. Respuesta correcta: ' + item.options[item.correct];
      fb.className = 'feedback show incorrect';
    }
  });

  const warning = document.getElementById('missingWarning');
  if(!allAnswered){
    warning.classList.add('show');
    document.getElementById('resultBox').classList.remove('show');
    return;
  }
  warning.classList.remove('show');

  const percent = Math.round((correctCount/10)*100);

  document.getElementById('scoreNum').textContent = correctCount + ' / 10';
  document.getElementById('scorePercent').textContent = percent + '% de respuestas correctas';

  let message = '';
  if(percent >= 90){
    message = 'Excelente dominio del modelo SCARF y su aplicación a la gestión educativa.';
  } else if(percent >= 70){
    message = 'Buen desempeño. Se recomienda reforzar los dominios señalados como incorrectos.';
  } else if(percent >= 50){
    message = 'Desempeño moderado. Se sugiere repasar los cinco dominios del modelo SCARF antes de continuar.';
  } else {
    message = 'Se recomienda revisar nuevamente los contenidos del submódulo antes de avanzar.';
  }
  document.getElementById('scoreMessage').textContent = message;

  document.getElementById('resultBox').classList.add('show');
  document.getElementById('resultBox').scrollIntoView({behavior:'smooth'});
});
